133 inline std::string FormatCommand(std::string_view tmpl, const std::unordered_map<std::string, std::string>& values) {
134  std::string result(tmpl);
135  for (const auto& [key, val] : values) {
136  std::string placeholder = "{" + key + "}";
137  size_t pos = 0;
138  while ((pos = result.find(placeholder, pos)) != std::string::npos) {
139  result.replace(pos, placeholder.length(), val);
140  pos += val.length();
141  }
142  }
143  return result;
144 }
