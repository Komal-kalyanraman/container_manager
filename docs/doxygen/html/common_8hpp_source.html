148 inline std::string FormatCommand(std::string_view tmpl, const std::unordered_map<std::string, std::string>& values) {
149  std::string result(tmpl);
150  for (const auto& [key, val] : values) {
151  std::string placeholder = "{" + key + "}";
152  size_t pos = 0;
153  while ((pos = result.find(placeholder, pos)) != std::string::npos) {
154  result.replace(pos, placeholder.length(), val);
155  pos += val.length();
156  }
157  }
158  return result;
159 }
